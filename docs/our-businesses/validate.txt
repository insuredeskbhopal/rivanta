async page => {
 const assert=(value,message)=>{if(!value)throw new Error(message);};const layouts=[],errors=[];page.on('pageerror',error=>errors.push(error.message));
 for(const width of [1920,1863,1440,1024,768,390,320]){
  await page.setViewportSize({width,height:900});await page.goto('http://localhost:3000/?business-check='+width+'#our-businesses');await page.waitForSelector('main[data-tools=ready]');await page.evaluate(()=>document.fonts.ready);await page.mouse.move(1,1);
  const section=page.locator('#our-businesses');await section.evaluate(el=>window.scrollTo({top:el.offsetTop-100,behavior:'instant'}));await section.locator('img').evaluateAll(images=>Promise.all(images.map(image=>image.decode())));
  const layout=await section.evaluate(el=>{const box=el.getBoundingClientRect();return{width:innerWidth,height:box.height,overflow:document.documentElement.scrollWidth>innerWidth,inside:[...el.querySelectorAll('h2,h3,p,a,img')].every(item=>{const r=item.getBoundingClientRect();return r.left>=box.left-1&&r.right<=box.right+1&&r.top>=box.top-1&&r.bottom<=box.bottom+1;}),headingClear:el.querySelector('h2').getBoundingClientRect().top>document.querySelector('header').getBoundingClientRect().bottom,headings:[...el.querySelectorAll('h3')].map(item=>item.textContent),visible:[...el.querySelectorAll('h3,p,a,img')].every(item=>item.checkVisibility()),images:[...el.querySelectorAll('img')].map(item=>({src:item.src,alt:item.alt,loaded:item.complete&&item.naturalWidth>0})),links:[...el.querySelectorAll('a')].map(item=>item.href),afterChallenges:!!(document.querySelector('#india-challenges').compareDocumentPosition(el)&Node.DOCUMENT_POSITION_FOLLOWING),disclosures:el.querySelectorAll('details,summary,button').length};});
  assert(!layout.overflow&&layout.inside&&layout.headingClear&&layout.visible&&layout.headings.length===4&&layout.images.length===4&&layout.images.every(item=>item.loaded)&&new Set(layout.images.map(item=>item.src)).size===4&&layout.afterChallenges&&layout.disclosures===0,'Layout '+JSON.stringify(layout));layouts.push(layout);
  await section.screenshot({path:'docs/our-businesses/section-'+width+'.png',style:'header,main > a[href="#company-overview"]{visibility:hidden !important}'});
 }
 await page.setViewportSize({width:1440,height:900});await page.goto('http://localhost:3000/?business-check=interactions#our-businesses');await page.waitForSelector('main[data-tools=ready]');
 const links=page.locator('#our-businesses a');await page.locator('#india-challenges a').last().focus();for(let i=0;i<4;i++){await page.keyboard.press('Tab');assert(await links.nth(i).evaluate(el=>el===document.activeElement&&el.matches(':focus-visible')),'Keyboard '+i);}
 await links.first().hover();await page.waitForTimeout(650);const hover=await page.locator('#our-businesses article img').first().evaluate(el=>getComputedStyle(el).transform);assert(hover.includes('1.025'),'Photo hover '+hover);
 await page.emulateMedia({reducedMotion:'reduce'});assert(await page.locator('#our-businesses article img').first().evaluate(el=>getComputedStyle(el).transform)==='none','Reduced image motion');assert(await links.first().locator('svg').evaluate(el=>getComputedStyle(el).transform)==='none','Reduced link motion');await page.emulateMedia({reducedMotion:'no-preference'});await page.mouse.move(1,1);
 const contrast=await page.locator('#our-businesses').evaluate(el=>{const lum=color=>color.match(/[0-9.]+/g).slice(0,3).map(Number).map(v=>v/255).map(v=>v<=.04045?v/12.92:((v+.055)/1.055)**2.4).reduce((a,v,i)=>a+v*[.2126,.7152,.0722][i],0);return Math.min(...[...el.querySelectorAll('h2,h3,p,a')].map(item=>{let bg=item;while(bg&&getComputedStyle(bg).backgroundColor==='rgba(0, 0, 0, 0)')bg=bg.parentElement;const a=lum(getComputedStyle(item).color),b=lum(getComputedStyle(bg).backgroundColor);return(Math.max(a,b)+.05)/(Math.min(a,b)+.05);}));});assert(contrast>=4.5,'Contrast '+contrast);
 await page.evaluate(()=>document.documentElement.style.zoom='2');assert(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth),'Zoom overflow');await page.evaluate(()=>document.documentElement.style.zoom='');
 await links.first().evaluate(el=>{window.scrollTo({top:el.getBoundingClientRect().top+scrollY-650,behavior:'instant'});el.focus({preventScroll:true});});await page.waitForTimeout(250);await page.screenshot({path:'docs/our-businesses/focus.png'});
 const y=await page.evaluate(()=>scrollY);await page.keyboard.press('PageUp');await page.waitForTimeout(400);assert(await page.evaluate(start=>scrollY<start,y),'Native scroll');
 const context=await page.context().browser().newContext({javaScriptEnabled:false,viewport:{width:390,height:844}});const nojs=await context.newPage();await nojs.goto('http://localhost:3000/#our-businesses');assert(await nojs.locator('#our-businesses article').evaluateAll(items=>items.length===4&&items.every(item=>item.checkVisibility())),'No-JavaScript content');await context.close();assert(errors.length===0,'Page errors '+JSON.stringify(errors));
 return{layouts,keyboard:true,hover,reducedMotion:true,minimumTextContrast:contrast,zoom200:true,nativeScrolling:true,noJavaScript:true,errors};
}
