async page => {
  await page.setViewportSize({width:390,height:844});await page.emulateMedia({reducedMotion:'reduce'});const response=await page.goto('http://127.0.0.1:3002/proof/cbg');await page.waitForSelector('main[data-tools="ready"]');
  const reduced=await page.evaluate(()=>({order:[...document.querySelectorAll('[data-plant]')].map(el=>el.dataset.plant),images:[...document.querySelectorAll('[data-stage] img')].map(img=>({complete:img.complete,transform:getComputedStyle(img.parentElement).transform})),descriptions:[...document.querySelectorAll('[id$="-description"]')].map(el=>el.id)}));
  if(reduced.order.join(',')!=='silicon,cbg,saf'||reduced.images.length!==4||reduced.images.some(i=>!i.complete||i.transform!=='none'))throw new Error('Reduced production fallback failed');await page.screenshot({path:'docs/phase-2/transition-proof/silicon-order/production-reduced.png',fullPage:true});
  const context=await page.context().browser().newContext({javaScriptEnabled:false,viewport:{width:390,height:844}});const fallback=await context.newPage();await fallback.goto('http://127.0.0.1:3002/proof/cbg');
  const noJavaScript={silicon:await fallback.locator('#silicon-description').isVisible(),sourceLink:await fallback.getByRole('link',{name:'Open the proposed Silicon concept image.'}).count(),disabled:await fallback.locator('button[data-jump]:disabled').count()};if(!noJavaScript.silicon||noJavaScript.sourceLink!==1||noJavaScript.disabled!==7)throw new Error('No-JavaScript Silicon fallback failed');await context.close();
  await page.emulateMedia({reducedMotion:'no-preference'});await page.route('**/_next/image?**',route=>route.request().url().includes('Silicon')?route.abort():route.continue());await page.goto('http://127.0.0.1:3002/proof/cbg');await page.locator('[data-error]').waitFor({state:'visible'});
  const failure={motion:await page.locator('main').getAttribute('data-motion'),description:await page.locator('#silicon-description').isVisible()};if(failure.motion||!failure.description)throw new Error('Image failure trapped content');await page.unroute('**/_next/image?**');
  return {status:response.status(),reduced,noJavaScript,imageFailure:failure};
}
