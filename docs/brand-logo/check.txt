async page=>{
 const results=[];
 for(const viewport of [{width:1440,height:900},{width:390,height:844},{width:320,height:568},{width:844,height:390},{width:768,height:1024}]){
  await page.setViewportSize(viewport);await page.emulateMedia({reducedMotion:'no-preference'});await page.goto('http://localhost:3000/proof/cbg');await page.waitForSelector('main[data-playback="overview"]');await page.locator('header img').evaluate(img=>img.decode());
  const result=await page.evaluate(()=>{const brand=document.querySelector('header > a'),img=brand.querySelector('img'),b=brand.getBoundingClientRect(),image=img.getBoundingClientRect(),nav=document.querySelector('header nav').getBoundingClientRect(),header=document.querySelector('header').getBoundingClientRect(),intro=document.querySelector('[data-intro]').getBoundingClientRect(),badge=document.querySelector('[data-concept]').getBoundingClientRect();return{width:innerWidth,height:innerHeight,source:img.currentSrc,complete:img.complete,naturalWidth:img.naturalWidth,brandText:brand.innerText,alt:img.alt,logoWidth:image.width,logoHeight:image.height,brandRight:b.right,navLeft:nav.left,navTop:nav.top,headerBottom:header.bottom,introTop:intro.top,captionGap:badge.top-intro.bottom,overflow:document.documentElement.scrollWidth>innerWidth};});
  const separateRows=result.navTop>=result.logoHeight+10;if(!result.complete||result.naturalWidth===0||!result.source.includes('logo-brand-lockup-light')||result.brandText.trim()!==''||result.overflow||result.introTop<result.headerBottom||result.captionGap<12||(!separateRows&&result.brandRight>result.navLeft))throw new Error('Logo layout failure '+JSON.stringify(result));results.push(result);await page.screenshot({path:`docs/brand-logo/header-${viewport.width}.png`});
 }
 await page.setViewportSize({width:320,height:568});await page.emulateMedia({reducedMotion:'reduce'});await page.goto('http://localhost:3000/proof/cbg');await page.waitForSelector('main[data-tools="ready"]');await page.locator('header img').evaluate(img=>img.decode());const reduced=await page.evaluate(()=>({headerBottom:document.querySelector('header').getBoundingClientRect().bottom,headlineTop:document.querySelector('h1').getBoundingClientRect().top,overflow:document.documentElement.scrollWidth>innerWidth,logo:document.querySelector('header img').complete}));if(reduced.overflow||reduced.headlineTop<reduced.headerBottom||!reduced.logo)throw new Error('Reduced logo layout failure');return{results,reduced};
}
