async page => {
 const check=(ok,message)=>{if(!ok)throw new Error(message);};const layouts=[];const errors=[];page.on('pageerror',error=>errors.push(error.message));
 for(const viewport of [{width:1920,height:1080},{width:1440,height:900},{width:1024,height:768},{width:768,height:1024},{width:390,height:844},{width:320,height:568}]){
  await page.setViewportSize(viewport);await page.goto('http://localhost:3000/?css-refinement='+viewport.width+'#india-challenges');await page.waitForSelector('main[data-tools=ready]');const section=page.locator('#india-challenges');await section.evaluate(el=>scrollTo(0,el.offsetTop-88));await section.locator('img').evaluateAll(images=>Promise.all(images.map(image=>image.decode())));
  const result=await section.evaluate(el=>{const box=el.getBoundingClientRect();return{width:innerWidth,height:box.height,overflow:document.documentElement.scrollWidth>innerWidth,inside:[...el.querySelectorAll('h2,h3,p,a,img')].every(item=>{const r=item.getBoundingClientRect();return r.left>=box.left-1&&r.right<=box.right+1&&r.top>=box.top-1&&r.bottom<=box.bottom+1;}),headingClear:el.querySelector('h2').getBoundingClientRect().top>document.querySelector('header').getBoundingClientRect().bottom,headings:[...el.querySelectorAll('[data-challenge] h3')].map(item=>item.textContent),paragraphs:el.querySelectorAll('[data-challenge] p').length,allVisible:[...el.querySelectorAll('[data-challenge] p')].every(item=>item.checkVisibility()),disclosures:el.querySelectorAll('details,summary').length,text:el.innerText,links:[...el.querySelectorAll('a')].map(item=>item.href)};});
  check(!result.overflow&&result.inside&&result.headingClear&&result.headings.length===8&&result.paragraphs===16&&result.allVisible&&result.disclosures===0,'Content layout failure '+JSON.stringify(result));check(result.text.includes('Growth That Creates Value At Home')&&result.text.includes('Cleaner Fuels For A Growing Nation')&&result.text.includes('below 99.99% purity')&&result.text.includes('commercial assessment')&&result.links.some(href=>href.includes('/2025/')&&href.endsWith('/280469')),'New content missing');delete result.text;layouts.push(result);
  await page.evaluate(()=>document.activeElement?.blur());await section.screenshot({path:'docs/india-challenges/css-refinement/section-'+viewport.width+'.png',style:'header,main > a[href="#company-overview"]{visibility:hidden !important}'});
 }
 await page.setViewportSize({width:1440,height:900});await page.goto('http://localhost:3000/?css-refinement=keyboard#india-challenges');await page.locator('#company-overview a').focus();for(let index=0;index<5;index++){await page.keyboard.press('Tab');check(await page.locator('#india-challenges a').nth(index).evaluate(el=>el===document.activeElement&&el.matches(':focus-visible')),'Keyboard link '+index);}
 await page.emulateMedia({reducedMotion:'reduce'});await page.locator('#india-challenges a').nth(1).hover();check(await page.locator('#india-challenges a').nth(1).evaluate(el=>getComputedStyle(el).transform)==='none','Reduced motion');await page.emulateMedia({reducedMotion:'no-preference'});
 const context=await page.context().browser().newContext({javaScriptEnabled:false,viewport:{width:390,height:844}});const nojs=await context.newPage();await nojs.goto('http://localhost:3000/#india-challenges');check(await nojs.locator('#india-challenges [data-challenge] p').evaluateAll(items=>items.length===16&&items.every(item=>item.checkVisibility())),'No-JS paragraphs');await context.close();check(errors.length===0,'Page errors '+JSON.stringify(errors));return{layouts,keyboard:true,reducedMotion:true,noJavaScript:true,errors};
}
